import com.sun.net.httpserver.*; import java.io.*; import java.net.*; public class WeatherWebService { public static void main(String[] args) throws Exception { HttpServer server = HttpServer.create(new InetSocketAddress(8000), 0); // WEATHER SERVICE server.createContext("/weather", (exchange) -> { String query = exchange.getRequestURI().getQuery(); String city = query.split("=")[1]; String weather = ""; // Simple Weather Database if (city.equalsIgnoreCase("Pune")) { weather = "25 Degree Celsius"; } else if (city.equalsIgnoreCase("Mumbai")) { weather = "30 Degree Celsius"; } else if (city.equalsIgnoreCase("Delhi")) { weather = "35 Degree Celsius"; } else if (city.equalsIgnoreCase("Nashik")) { weather = "24 Degree Celsius"; } else { weather = "Weather Data Not Available"; } String response = "" + "

Weather Report

" + "City = " + city + "

" + "Temperature = " + weather + "

" + "Go Back" + ""; exchange.sendResponseHeaders(200, response.length()); OutputStream os = exchange.getResponseBody(); os.write(response.getBytes()); os.close(); }); server.start(); System.out.println("Server running at http://localhost:8000"); } }
